## 一、防抖（Debounce）

**场景：** 搜索输入框，用户停止输入 300ms 后才发请求。

```javascript
function debounce(fn, delay = 300, immediate = false) {
  let timer = null

  return function (...args) {
    const context = this

    // immediate 为 true 时立即执行第一次
    if (immediate && !timer) {
      fn.apply(context, args)
    }

    clearTimeout(timer)
    timer = setTimeout(() => {
      fn.apply(context, args)
      timer = null
    }, delay)
  }
}

// 测试
const log = debounce((msg) => console.log(msg), 500)
log('A')  // 500ms 内多次调用
log('B')
log('C')  // 500ms 后只输出 'C'
```

**考察要点：** `this` 指向、闭包、`setTimeout`、`apply`/`call`。

## 二、节流（Throttle）

**场景：** 滚动事件、拖拽，固定频率触发。

```javascript
function throttle(fn, interval = 100) {
  let lastTime = 0

  return function (...args) {
    const context = this
    const now = Date.now()

    if (now - lastTime >= interval) {
      lastTime = now
      fn.apply(context, args)
    }
  }
}

// 测试
const scrollHandler = throttle(() => console.log('滚动中'), 200)
window.addEventListener('scroll', scrollHandler)
// 每秒最多触发 5 次（而不是几十次）
```

**防抖 vs 节流：**

| 场景 | 方案 | 说明 |
|------|------|------|
| 搜索输入 | **防抖** | 停止输入后才触发 |
| 滚动/拖拽 | **节流** | 固定频率触发 |
| 按钮点击 | **防抖** | 防止重复提交 |

## 三、深拷贝（Deep Clone）

```javascript
function deepClone(value, cache = new WeakMap()) {
  // 基本类型直接返回
  if (value === null || typeof value !== 'object') return value

  // 处理循环引用
  if (cache.has(value)) return cache.get(value)

  // 处理特殊对象
  if (value instanceof Date) return new Date(value)
  if (value instanceof RegExp) return new RegExp(value.source, value.flags)
  if (value instanceof Map) {
    const result = new Map()
    cache.set(value, result)
    value.forEach((v, k) => result.set(k, deepClone(v, cache)))
    return result
  }
  if (value instanceof Set) {
    const result = new Set()
    cache.set(value, result)
    value.forEach((v) => result.add(deepClone(v, cache)))
    return result
  }

  // 数组或对象
  const result = Array.isArray(value) ? [] : {}
  cache.set(value, result)

  Reflect.ownKeys(value).forEach((key) => {
    result[key] = deepClone(value[key], cache)
  })

  return result
}

// 测试
const obj = { a: 1, b: { c: 2 }, d: [1, 2, 3] }
obj.self = obj  // 循环引用

const cloned = deepClone(obj)
console.log(cloned)           // { a: 1, b: { c: 2 }, d: [1, 2, 3], self: [Circular] }
console.log(cloned.b === obj.b) // false（深拷贝）
```

**为什么不直接用 `JSON.parse(JSON.stringify(obj))`？**
- 不支持函数、undefined、Symbol、BigInt
- 不支持 Date（变为字符串）
- 不支持循环引用

## 四、数组去重

```javascript
// 方式一：Set（最简单）
function unique(arr) {
  return [...new Set(arr)]
}

// 方式二：filter + indexOf
function unique2(arr) {
  return arr.filter((item, index) => arr.indexOf(item) === index)
}

// 方式三：reduce
function unique3(arr) {
  return arr.reduce((acc, cur) => {
    if (!acc.includes(cur)) acc.push(cur)
    return acc
  }, [])
}

// 方式四：对象 key（适合大量数据）
function unique4(arr) {
  const map = {}
  return arr.filter((item) => {
    const key = typeof item + JSON.stringify(item)
    if (map[key]) return false
    map[key] = true
    return true
  })
}

// 测试
console.log(unique([1, 2, 2, 3, 3, 4])) // [1, 2, 3, 4]
console.log(unique([{a:1}, {a:1}, {a:2}])) // 对象去重需用方式四
```

## 五、手写 Promise.all

```javascript
Promise.myAll = function (promises) {
  return new Promise((resolve, reject) => {
    if (!Array.isArray(promises)) {
      return reject(new TypeError('参数必须是数组'))
    }

    const results = []
    let completed = 0

    if (promises.length === 0) {
      return resolve(results)
    }

    promises.forEach((promise, index) => {
      Promise.resolve(promise).then((value) => {
        results[index] = value
        completed++

        if (completed === promises.length) {
          resolve(results)
        }
      }).catch(reject)
    })
  })
}

// 测试
const p1 = Promise.resolve(1)
const p2 = Promise.resolve(2)
const p3 = new Promise((r) => setTimeout(() => r(3), 100))

Promise.myAll([p1, p2, p3]).then(console.log)  // [1, 2, 3]

// 有一个失败则整体失败
Promise.myAll([p1, Promise.reject('err'), p3])
  .catch(console.error)  // 'err'
```

## 六、手写 Promise.race

```javascript
Promise.myRace = function (promises) {
  return new Promise((resolve, reject) => {
    if (!Array.isArray(promises)) {
      return reject(new TypeError('参数必须是数组'))
    }

    promises.forEach((promise) => {
      Promise.resolve(promise).then(resolve).catch(reject)
    })
  })
}

// 测试
const slow = new Promise((r) => setTimeout(() => r('慢'), 200))
const fast = new Promise((r) => setTimeout(() => r('快'), 100))

Promise.myRace([slow, fast]).then(console.log)  // '快'
```

## 七、发布订阅模式（Event Emitter）

```javascript
class EventEmitter {
  constructor() {
    this.events = new Map()
  }

  on(event, callback) {
    if (!this.events.has(event)) {
      this.events.set(event, new Set())
    }
    this.events.get(event).add(callback)
    return this
  }

  off(event, callback) {
    if (!this.events.has(event)) return
    this.events.get(event).delete(callback)
    return this
  }

  emit(event, ...args) {
    if (!this.events.has(event)) return
    this.events.get(event).forEach((cb) => cb(...args))
  }

  once(event, callback) {
    const wrapper = (...args) => {
      callback(...args)
      this.off(event, wrapper)
    }
    this.on(event, wrapper)
    return this
  }
}

// 测试
const bus = new EventEmitter()
const handler = (msg) => console.log('收到:', msg)

bus.on('message', handler)
bus.emit('message', '你好')  // 收到: 你好
bus.off('message', handler)
bus.emit('message', '再见')  // 不输出（已取消订阅）

bus.once('once', () => console.log('只执行一次'))
bus.emit('once')  // 只执行一次
bus.emit('once')  // 不再执行
```

## 八、函数柯里化（Currying）

```javascript
function curry(fn, ...args) {
  // 如果传入的参数个数 >= 原函数参数个数，直接执行
  if (args.length >= fn.length) {
    return fn(...args)
  }

  // 否则返回一个新函数，继续收集参数
  return function (...args2) {
    return curry(fn, ...args, ...args2)
  }
}

// 测试
function add(a, b, c) {
  return a + b + c
}

const curriedAdd = curry(add)
console.log(curriedAdd(1)(2)(3))   // 6
console.log(curriedAdd(1, 2)(3))   // 6
console.log(curriedAdd(1)(2, 3))   // 6
```

## 九、手写 call / apply / bind

```javascript
// call：立即执行，参数列表
Function.prototype.myCall = function (context, ...args) {
  context = context ?? window
  const key = Symbol('fn')
  context[key] = this
  const result = context[key](...args)
  delete context[key]
  return result
}

// apply：立即执行，参数数组
Function.prototype.myApply = function (context, args) {
  context = context ?? window
  const key = Symbol('fn')
  context[key] = this
  const result = context[key](...(args || []))
  delete context[key]
  return result
}

// bind：返回新函数，不立即执行
Function.prototype.myBind = function (context, ...args) {
  const fn = this
  return function (...args2) {
    return fn.myApply(context, [...args, ...args2])
  }
}

// 测试
const obj = { name: '测试' }
function greet(greeting, punctuation) {
  console.log(`${greeting}, ${this.name}${punctuation}`)
}

greet.myCall(obj, '你好', '！')       // 你好, 测试！
greet.myApply(obj, ['你好', '！'])    // 你好, 测试！
const bound = greet.myBind(obj, '你好')
bound('！')                           // 你好, 测试！
```

## 十、手写 instanceof

```javascript
function myInstanceof(left, right) {
  let proto = Object.getPrototypeOf(left)

  while (true) {
    if (proto === null) return false
    if (proto === right.prototype) return true
    proto = Object.getPrototypeOf(proto)
  }
}

// 测试
class Animal {}
const dog = new Animal()

console.log(myInstanceof(dog, Animal))     // true
console.log(myInstanceof(dog, Object))     // true
console.log(myInstanceof(dog, Array))      // false
console.log(myInstanceof([], Array))       // true
```
